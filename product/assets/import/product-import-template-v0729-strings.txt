36	企业安全信息核验接口，依托企业主体数据，支持统一社会信用代码查询，快速识别经营异常、失信等风险标签，输出风险等级，高效满足业务准入安全筛查需求。
37	API
38	1990/01/01 -
39	全国
40	每日
41	openapi: 3.0.3
info:
  title: 企业安全信息核验接口
  description: 企业主体安全信息实时核验服务
  version: 1.0.0
servers:
  - url: https://api.example.com/v1
paths:
  /enterprise/security/verify:
    post:
      summary: 企业安全信息核验
      description: 根据统一社会信用代码核验企业安全风险相关信息
      requestBody:
        required: true
        content:
          application/json:
            schema:
              $ref: '#/components/schemas/VerifyRequest'
      responses:
        '200':
          description: 查询成功
          content:
            application/json:
              schema:
                $ref: '#/components/schemas/VerifyResponse'
        '400':
          description: 请求参数错误
        '401':
          description: 鉴权失败
        '403':
          description: 接口权限不足
components:
  securitySchemes:
    ApiKeyAuth:
      type: apiKey
      in: header
      name: X-API-Key
  security:
    - ApiKeyAuth: []
  schemas:
    VerifyRequest:
      type: object
      required:
        - uscc
      properties:
        uscc:
          type: string
          description: 统一社会信用代码
          example: 91310101XXXXXXXXXX
        name:
          type: string
          description: 企业名称（可选，辅助校验）
          example: XX科技有限公司
    VerifyResponse:
      type: object
      properties:
        code:
          type: integer
          description: 返回码 0成功
          example: 0
        msg:
          type: string
          description: 返回信息
          example: success
        data:
          $ref: '#/components/schemas/EnterpriseSecurityInfo'
    EnterpriseSecurityInfo:
      type: object
      properties:
        uscc:
          type: string
          description: 统一社会信用代码
        enterpriseName:
          type: string
          description: 企业名称
        riskLevel:
          type: string
          enum: [SAFE, WARNING, DANGER]
          description: 风险等级 安全/预警/高危
        riskTags:
          type: array
          items:
            type: string
          description: 风险标签列表
          example: ["行政处罚","经营异常"]
        valid:
          type: boolean
          description: 企业主体是否有效
42	msg string 返回提示信息
data object 核验结果数据
uscc string 统一社会信用代码
enterpriseName string 企业名称
